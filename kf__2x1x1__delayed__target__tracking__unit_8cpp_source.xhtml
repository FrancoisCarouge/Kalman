98constexpr auto time_step{0.02 * s};

167[[maybe_unused]] const auto sample{[] -> int {
168 constexpr int measurement_period{5}; // predictions
169 constexpr int measurement_delay{5}; // predictions
170 const auto measurement_deviation{0.1 * m};
171 const auto acceleration_deviation{0.05 * m / s2};
172 // The target's velocity random walk, its variance growth rate.
173 const auto target_velocity_diffusion{0.01 * m2 / s3};
174 // The vehicle's sway: its acceleration amplitude and angular frequency.
175 const auto sway_acceleration{2. * m / s2};
176 const auto sway_frequency{0.5 * rad / s};
177
178 // The seed of the sensor noise, from the system's entropy source: a
179 // blocking input-output operation, out of the real-time verification.
180 const auto seed{[] -> std::random_device::result_type {
181 const not_realtime opt_out;
182 return std::random_device{}();
183 }()};
184 // The sensor noise, a standard normal sequence. The plots are of the seed 1,
185 // with the GNU C++ standard library: the algorithm of the distribution, so
186 // the sequence of a seed, varies across standard libraries.
187 std::minstd_rand engine{seed};
188 std::normal_distribution distribution{0., 1.};
189 const auto noise{[&] -> double { return distribution(engine); }};
190
191 // The constant velocity model driven by the vehicle's acceleration: the
192 // relative acceleration is its opposite. The process noise is the
193 // accelerometer's, integrated over a step, and the target's velocity
194 // random walk.
195 const auto acceleration_variance{pow<2>(acceleration_deviation)};
196 const auto make_filter{[&] -> auto {
197 return kalman{
198 // The state X: the relative position [m] and velocity [m.s^-1].
199 state{track{position{10. * m}, velocity{0. * m / s}}},
200 // The output Z: the measured relative position [m].
201 output<sighting>,
202 // The input U: the vehicle's measured acceleration [m.s^-2].
203 input<acceleration>,
204 // The estimate uncertainty P [m^2, m^2.s^-2]:
205 estimate_uncertainty{[]() -> track_covariance {
206 track_covariance value;
207 value.at<0, 0>(4. * m2);
208 value.at<1, 1>(4. * m2 / s2);
209 return value;
210 }()},
211 // The process uncertainty Q:
212 process_uncertainty{[&]() -> track_covariance {
213 track_covariance value;
214 value.at<0, 0>(acceleration_variance * pow<4>(time_step) / 4.);
215 value.at<0, 1>(acceleration_variance * pow<3>(time_step) / 2.);
216 value.at<1, 0>(acceleration_variance * pow<3>(time_step) / 2.);
217 value.at<1, 1>((acceleration_variance * pow<2>(time_step)) +
218 (target_velocity_diffusion * time_step));
219 return value;
220 }()},
221 // The output uncertainty R [m^2]:
222 output_uncertainty{pow<2>(measurement_deviation)},
223 // The output model H:
224 output_model{[]() -> sighting_model {
225 sighting_model value;
226 value.at<0>(1.);
227 return value;
228 }()},
229 // The state transition F:
230 state_transition{[]() -> track_transition {
231 track_transition value{track_transition::matrix::Identity()};
232 value.at<0, 1>(time_step);
233 return value;
234 }()},
235 // The input control G: the relative motion is opposite the vehicle's.
236 input_control{[]() -> acceleration_control {
237 acceleration_control value;
238 value.at<0>(-pow<2>(time_step) / 2.);
239 value.at<1>(-time_step);
240 return value;
241 }()}};
242 }};
243
244 // The filter fusing each measurement as it arrives, and the one rewinding to
245 // its instant.
246 auto late{make_filter()};
247 auto rewound{make_filter()};
248
249 // The history of the rewound filter: the track and estimate uncertainty
250 // after each of the last predictions, and the input of each.
251 struct snapshot {
252 track estimate;
253 track_covariance covariance;
254 acceleration measured_acceleration;
255 };
256 // The snapshots of a measurement's instant through the current one.
257 constexpr std::size_t history_size{measurement_delay + 1};
258 std::array<snapshot, history_size> history{};
259
260 // The camera's measurements in flight, the relative positions of the
261 // instants they describe, within a delay through the current one.
262 constexpr std::size_t in_flight_size{
263 (measurement_delay / measurement_period) + 1};
264 std::array<position, in_flight_size> in_flight{};
265
266 position true_position{10. * m};
267 velocity target_velocity{0. * m / s};
268 squared_position late_squared_errors{0. * m2};
269 squared_position rewound_squared_errors{0. * m2};
270 int fused{0};
271 const int steps{3000};
272 const int settled{250};
273 for (int step{1}; step <= steps; ++step) {
274 const auto phase{sway_frequency * (step * time_step)};
275 const acceleration vehicle_acceleration{sway_acceleration *
276 mp_units::si::sin(phase)};
277 target_velocity += sqrt(target_velocity_diffusion * time_step) * noise();
278 const velocity true_velocity{target_velocity -
279 (sway_acceleration / sway_frequency * rad *
280 (1. - mp_units::si::cos(phase)))};
281 true_position += true_velocity * time_step;
282 const acceleration measured_acceleration{
283 vehicle_acceleration + (acceleration_deviation * noise())};
284
285 late.predict(measured_acceleration);
286 rewound.predict(measured_acceleration);
287 history.at(static_cast<std::size_t>(step) %
288 history_size) = {.estimate = rewound.x(),
289 .covariance = rewound.p(),
290 .measured_acceleration = measured_acceleration};
291
292 // The camera describes this instant, and delivers it later.
293 if (step % measurement_period == 0) {
294 in_flight.at(static_cast<std::size_t>(step / measurement_period) %
295 in_flight_size) =
296 true_position + (measurement_deviation * noise());
297 }
298
299 // A measurement arrives, describing the step one delay earlier.
300 if (const int described{step - measurement_delay};
301 described > 0 && described % measurement_period == 0) {
302 const position arrived{in_flight.at(
303 static_cast<std::size_t>(described / measurement_period) %
304 in_flight_size)};
305
306 late.update(arrived);
307
308 // Rewind to the measurement's instant, update there, then replay the
309 // predictions since, refreshing their history.
310 const snapshot &then{
311 history.at(static_cast<std::size_t>(described) % history_size)};
312 rewound.x(then.estimate);
313 rewound.p(then.covariance);
314 rewound.update(arrived);
315 for (int replayed{described + 1}; replayed <= step; ++replayed) {
316 snapshot &entry{
317 history.at(static_cast<std::size_t>(replayed) % history_size)};
318 rewound.predict(entry.measured_acceleration);
319 entry.estimate = rewound.x();
320 entry.covariance = rewound.p();
321 }
322 ++fused;
323 }
324
325 // After the first seconds, the error of each estimate.
326 if (step > settled) {
327 late_squared_errors += pow<2>(late.x().at<0>() - true_position);
328 rewound_squared_errors += pow<2>(rewound.x().at<0>() - true_position);
329 }
330 }
331
332 // The root mean square errors of the estimates.
333 const position late_error{sqrt(late_squared_errors / (steps - settled))};
334 const position rewound_error{
335 sqrt(rewound_squared_errors / (steps - settled))};
336
337 // The rewound filter tracks the target within the measurement noise; the
338 // late one lags it, whatever the seed of the noise.
339 assert(fused == 599);
340 assert(rewound_error < 0.1 * m &&
341 "The rewound estimate stays near the target.");
342 assert(late_error > 4. * rewound_error &&
343 "Fusing a delayed measurement as current lags the target.");
344
345 return 0;
346}()};
