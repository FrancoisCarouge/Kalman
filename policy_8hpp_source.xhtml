105// The policies of a filter, in their canonical order: the update form, the
106// symmetrization, then the floors. The filter holds its policies as its first
107// member, without a unique address, which stores nothing but the floors with a
108// runtime standard deviation: a policy declared or not is part of the filter
109// type, at no space cost, and its absence selects the default filter.
110template <typename... Policies> struct policies {};

112template <typename... Policies>
113 requires(stored_floors<Policies> || ...)
114struct policies<Policies...> {
115 decltype(std::tuple_element_t<sizeof...(Policies) - 1,
116 std::tuple<Policies...>>::value) deviations;
117};

176// A matrix whose elements are accessed at compile-time positions, by the
177// multidimensional subscript or the function call operators: no backend
178// customization point, and heterogeneous typed matrices are supported.
179template <typename Type>
180concept indexable = requires(Type value) { value[position<0>, position<0>]; } ||
181 requires(Type value) { value(position<0>, position<0>); };

183template <std::size_t Row, std::size_t Column>
184[[nodiscard]] constexpr decltype(auto) at(auto &value) {
185 if constexpr (requires { value[position<Row>, position<Column>]; }) {
186 return value[position<Row>, position<Column>];
187 } else {
188 return value(position<Row>, position<Column>);

192// Averages the estimate uncertainty with its transpose. A scalar is symmetric.
193template <typename Type> constexpr void symmetrize_estimate(Type &p) {
194 if constexpr (indexable<Type>) {
195 using element = std::remove_cvref_t<decltype(at<0, 0>(p))>;
196
197 if constexpr (std::floating_point<element>) {
198 p = Type{(p + t(p)) * element{0.5}};
199 } else {
200 p = Type{(p + t(p)) * 0.5};
201 }
202 }
203}

205// Raises the variance to the square of its standard deviation floor.
206constexpr void raise_variance(auto &variance, const auto &deviation) {
207 const auto floor{deviation * deviation};
208
209 if (variance < floor) {
210 variance = floor;
211 }
212}

214// Raises each diagonal variance of the estimate uncertainty, in state order.
215template <typename Type, typename... Deviations>
216constexpr void raise_variances(Type &p,
217 const std::tuple<Deviations...> &deviations) {
218 if constexpr (indexable<Type>) {
219 static_assert(std::is_lvalue_reference_v<decltype(at<0, 0>(p))>,
220 "The floors require assignable estimate uncertainty "
221 "elements from the linear algebra backend.");
222
223 [&]<std::size_t... Indexes>(std::index_sequence<Indexes...>) {
224 (raise_variance(at<Indexes, Indexes>(p),
225 deviation_of(std::get<Indexes>(deviations))),
226 ...);
227 }(std::index_sequence_for<Deviations...>{});
228 } else {
229 raise_variance(p, deviation_of(std::get<0>(deviations)));
230 }
231}

233// Conditions the estimate uncertainty after an update or a prediction.
234template <typename... Policies>
235constexpr void condition(const policies<Policies...> &configuration, auto &p) {
236 if constexpr ((std::same_as<Policies, symmetrize_t> || ...)) {
237 symmetrize_estimate(p);
238 }
239
240 if constexpr (sizeof...(Policies) > 0) {
241 using last =
242 std::tuple_element_t<sizeof...(Policies) - 1, std::tuple<Policies...>>;
243
244 if constexpr (stored_floors<last>) {
245 raise_variances(p, configuration.deviations);
246 } else if constexpr (floors_policy<last>) {
247 raise_variances(p, decltype(last::value){});
248 }
249 }
250}

252// Updates the estimate uncertainty of a filter with an output model H, then
253// conditions it. The forms agree in exact arithmetic, the gain being optimal.
254// The simple form of a scalar output and a matrix state is associated as
255// P - K(HP), and the default Joseph form as (P - K(HP)) - (P - K(HP))HᵀKᵀ +
256// KRKᵀ: the same products, at a quadratic rather than a cubic cost in the state
257// dimension.
258template <typename... Policies>
259constexpr void
260update_estimate_uncertainty(const policies<Policies...> &configuration, auto &p,
261 const auto &i, const auto &k, const auto &h,
262 const auto &r, const auto &s) {
263 using estimate_uncertainty = std::remove_cvref_t<decltype(p)>;
264 using kinds = policies<Policies...>;
265
266 constexpr bool scalar_output{indexable<estimate_uncertainty> &&
267 not indexable<std::remove_cvref_t<decltype(s)>>};
268
269 if constexpr (contains<simple_form_t, kinds> && scalar_output) {
270 p = estimate_uncertainty{p - k * (h * p)};
271 } else if constexpr (contains<simple_form_t, kinds>) {
272 p = estimate_uncertainty{(i - k * h) * p};
273 } else if constexpr (contains<symmetric_form_t, kinds>) {
274 p = estimate_uncertainty{p - k * s * t(k)};
275 } else if constexpr (contains<information_form_t, kinds>) {
276 p = estimate_uncertainty{i / (i / p + t(h) / r * h)};
277 } else if constexpr (scalar_output) {
278 p = estimate_uncertainty{p - k * (h * p)};
279 p = estimate_uncertainty{p - p * t(h) * t(k) + k * r * t(k)};
280 } else {
281 p = estimate_uncertainty{(i - k * h) * p * t(i - k * h) + k * r * t(k)};
282 }
283
284 condition(configuration, p);
285}

287// Updates the estimate uncertainty of a filter without an output model, the
288// identity, then conditions it.
289template <typename... Policies>
290constexpr void
291update_estimate_uncertainty(const policies<Policies...> &configuration, auto &p,
292 const auto &i, const auto &k, const auto &r,
293 const auto &s) {
294 using estimate_uncertainty = std::remove_cvref_t<decltype(p)>;
295 using kinds = policies<Policies...>;
296
297 if constexpr (contains<simple_form_t, kinds>) {
298 p = estimate_uncertainty{(i - k) * p};
299 } else if constexpr (contains<symmetric_form_t, kinds>) {
300 p = estimate_uncertainty{p - k * s * t(k)};
301 } else if constexpr (contains<information_form_t, kinds>) {
302 p = estimate_uncertainty{i / (i / p + i / r)};
303 } else {
304 p = estimate_uncertainty{(i - k) * p * t(i - k) + k * r * t(k)};
305 }
306
307 condition(configuration, p);
308}
