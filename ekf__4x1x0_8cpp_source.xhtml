48template <auto Policy>
49const auto step{[](float drift_x, float drift_y, float position_x,
50 float position_y, float lift) -> void {
51 static kalman filter{
52 state{vector<4>{2.F, 80.F, 5.F, 5.F}},
53 output<vector<1>>,
54 estimate_uncertainty{{0.0049F, 0.F, 0.F, 0.F},
55 {0.F, 400.F, 0.F, 0.F},
56 {0.F, 0.F, 400.F, 0.F},
57 {0.F, 0.F, 0.F, 400.F}},
58 process_uncertainty{{0.0001F, 0.F, 0.F, 0.F},
59 {0.F, 0.0001F, 0.F, 0.F},
60 {0.F, 0.F, 0.0001F, 0.F},
61 {0.F, 0.F, 0.F, 0.0001F}},
62 output_uncertainty{0.2025F},
63 output_model{[](const vector<4> &x, const float &px,
64 const float &py) -> matrix<1, 4> {
65 return matrix<1, 4>{x[1], x[0], -2.F * (x[2] - px), -2.F * (x[3] - py)};
66 }},
67 transition{[](const vector<4> &x, const float &dx, const float &dy)
68 -> vector<4> { return x + vector<4>{0.F, 0.F, dx, dy}; }},
69 observation{[](const vector<4> &x, const float &px,
70 const float &py) -> vector<1> {
71 return vector<1>{(x[0] * x[1]) - ((x[2] - px) * (x[2] - px)) -
72 ((x[3] - py) * (x[3] - py))};
73 }},
74 update_types<float, float>,
75 prediction_types<float, float>,
76 Policy};
77
78 filter.predict(drift_x, drift_y);
79 filter.update(position_x, position_y, lift);
80}};
81} // namespace
82} // namespace fcarouge::benchmark
