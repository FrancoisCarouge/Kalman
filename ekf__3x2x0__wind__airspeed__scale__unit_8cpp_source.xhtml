179const auto air_velocity_of{
180 [](const air_data &estimate, const velocity &ground) -> velocity {
181 return ground - velocity{estimate.at<0>(), estimate.at<1>(),
182 down_velocity(0. * m / s)};
183 }};

187struct body_air_velocity {
188 quantity<forward_velocity[m / s]> forward;
189 quantity<right_velocity[m / s]> right;
190};

199const auto to_body{
200 [](const velocity &air, const heading &nose) -> body_air_velocity {
201 const auto [north, east, down]{air};
202 const speed along_north{north};
203 const speed along_east{east};
204 const auto cosine{mp_units::si::cos(nose)};
205 const auto sine{mp_units::si::sin(nose)};
206 return {.forward = forward_velocity(
207 speed{(cosine * along_north) + (sine * along_east)}),
208 .right = right_velocity(
209 speed{(-sine * along_north) + (cosine * along_east)})};
210 }};

215const auto observe{[](const air_data &estimate, const velocity &ground,
216 const heading &nose) -> reading {
217 const velocity air{air_velocity_of(estimate, ground)};
218 const airspeed true_speed{true_airspeed(air.magnitude())};
219 const body_air_velocity body{to_body(air, nose)};
220 return reading{estimate.at<2>() * true_speed, body.right / body.forward};
221}};
222
231const auto observation_jacobian{[](const air_data &estimate,
232 const velocity &ground,
233 const heading &nose) -> reading_jacobian {
234 const velocity air{air_velocity_of(estimate, ground)};
235 const airspeed true_speed{true_airspeed(air.magnitude())};
236 const body_air_velocity body{to_body(air, nose)};
237 const auto [north, east, down]{air};
238 const speed magnitude{true_speed};
239 const speed along_north{north};
240 const speed along_east{east};
241 const speed forward{body.forward};
242 const speed right{body.right};
243 const auto cosine{mp_units::si::cos(nose)};
244 const auto sine{mp_units::si::sin(nose)};
245 const auto scale{estimate.at<2>()};
246 reading_jacobian value;
247 value.at<0, 0>((indicated_airspeed /
248 north_velocity)(-speed{scale * along_north} / magnitude));
249 value.at<0, 1>((indicated_airspeed /
250 east_velocity)(-speed{scale * along_east} / magnitude));
251 value.at<0, 2>(reading_jacobian::element<0, 2>{true_speed});
252 value.at<1, 0>((sideslip / north_velocity)(
253 ((sine * forward) + (cosine * right)) / (forward * forward)));
254 value.at<1, 1>((sideslip / east_velocity)(
255 ((-cosine * forward) + (sine * right)) / (forward * forward)));
256 return value;
257}};

329[[maybe_unused]] const auto sample{[] -> int {
330 const auto time_step{0.1 * s};
331 const auto turn_rate{0.1 * rad / s};
332 const airspeed true_speed{20. * true_airspeed[m / s]};
333 const quantity<north_velocity[m / s]> wind_north{north_velocity(4. * m / s)};
334 const quantity<east_velocity[m / s]> wind_east{east_velocity(3. * m / s)};
335 const scale_factor scale{0.9 * airspeed_scale[mp_units::one]};
336 const auto spike_height{30. * indicated_airspeed[m / s]};
337 const auto airspeed_deviation{0.5 * indicated_airspeed[m / s]};
338 const auto sideslip_deviation{0.05 * sideslip[mp_units::one]};
339 // The random walks of the wind, 0.01 m.s^-1, and of the scale factor, 1e-4,
340 // per square root of second: their variance growth rates.
341 const auto wind_diffusion{0.01 * 0.01 * m2 / s3};
342 const auto scale_diffusion{1e-4 * 1e-4 / s};
343 const double gate{5.}; // standard deviations
344
345 // The seed of the sensor noise, from the system's entropy source: a
346 // blocking input-output operation, out of the real-time verification.
347 const auto seed{[] -> std::random_device::result_type {
348 const not_realtime opt_out;
349 return std::random_device{}();
350 }()};
351 // The sensor noise, a standard normal sequence. The plots are of the seed 1,
352 // with the GNU C++ standard library: the algorithm of the distribution, so
353 // the sequence of a seed, varies across standard libraries.
354 std::minstd_rand engine{seed};
355 std::normal_distribution distribution{0., 1.};
356 const auto noise{[&] -> double { return distribution(engine); }};
357
358 kalman filter{
359 // The state X: no wind and a unit scale factor.
360 state{air_data{north_velocity(0. * m / s), east_velocity(0. * m / s),
361 1. * airspeed_scale[mp_units::one]}},
362 // The output Z: the indicated airspeed [m.s^-1] and the sideslip [1].
363 output<reading>,
364 // The estimate uncertainty P:
365 estimate_uncertainty{[]() -> air_data_covariance {
366 air_data_covariance value;
367 value.at<0, 0>(25. * m2 / s2);
368 value.at<1, 1>(25. * m2 / s2);
369 value.at<2, 2>(0.1 * 0.1 * mp_units::one);
370 return value;
371 }()},
372 // The process uncertainty Q, of the random walks over a step:
373 process_uncertainty{[&]() -> air_data_covariance {
374 air_data_covariance value;
375 value.at<0, 0>(wind_diffusion * time_step);
376 value.at<1, 1>(wind_diffusion * time_step);
377 value.at<2, 2>(scale_diffusion * time_step);
378 return value;
379 }()},
380 // The output uncertainty R:
381 output_uncertainty{[&]() -> reading_covariance {
382 reading_covariance value;
383 value.at<0, 0>(airspeed_deviation * airspeed_deviation);
384 value.at<1, 1>(sideslip_deviation * sideslip_deviation);
385 return value;
386 }()},
387 // The output model H = ∂h/∂X, of the ground velocity and the heading:
388 output_model{observation_jacobian},
389 // The wind and the scale factor are constant: their prediction is their
390 // last estimate, so no transition function is declared.
391 // The observation h(X) of the airspeed and the sideslip:
392 observation{observe},
393 // The filter update uses the ground velocity and the heading.
394 update_types<velocity, heading>};
395
396 // The consistency test of the observations, computed outside of the filter:
397 // each innovation within its gate.
398 const auto consistent{[&filter, gate](const velocity &ground,
399 const heading &nose,
400 const reading &measured) -> bool {
401 const reading innovation{measured - observe(filter.x(), ground, nose)};
402 const reading_jacobian jacobian{
403 observation_jacobian(filter.x(), ground, nose)};
404 const reading_covariance innovation_uncertainty{
405 (jacobian * filter.p() * transposed(jacobian)) + filter.r()};
406 return innovation.at<0>() * innovation.at<0>() <
407 gate * gate * innovation_uncertainty.at<0, 0>() &&
408 innovation.at<1>() * innovation.at<1>() <
409 gate * gate * innovation_uncertainty.at<1, 1>();
410 }};
411
412 int spikes_fused{0};
413 int consistent_rejected{0};
414 heading nose{0. * rad};
415 const int steps{3000};
416 for (int step{0}; step < steps; ++step) {
417 // The heading is that of the air velocity: the flight is coordinated.
418 const speed air_speed{true_speed};
419 const velocity ground{
420 north_velocity(air_speed * mp_units::si::cos(nose)) + wind_north,
421 east_velocity(air_speed * mp_units::si::sin(nose)) + wind_east,
422 down_velocity(0. * m / s)};
423 // Every tenth step, an airspeed corrupted by a spike.
424 const bool spike{step % 10 == 5};
425 const reading measured{
426 indicated_airspeed(scale * true_speed) +
427 (airspeed_deviation * noise()) +
428 (spike ? spike_height : 0. * indicated_airspeed[m / s]),
429 sideslip_deviation * noise()};
430
431 filter.predict();
432
433 if (consistent(ground, nose, measured)) {
434 filter.update(ground, nose, measured);
435
436 // Keep the scale factor within its bounds.
437 air_data bounded{filter.x()};
438 bounded.at<2>(std::clamp(bounded.at<2>(),
439 0.5 * airspeed_scale[mp_units::one],
440 1.5 * airspeed_scale[mp_units::one]));
441 filter.x(bounded);
442
443 spikes_fused += static_cast<int>(spike);
444 } else {
445 consistent_rejected += static_cast<int>(!spike);
446 }
447
448 nose += turn_rate * time_step;
449 }
450
451 // The wind and the scale factor are found, the spikes rejected, whatever the
452 // seed of the noise. At five standard deviations, a consistent update is
453 // rejected about once in three hundred runs.
454 assert(abs(filter.x().at<0>() - wind_north) < 0.25 * m / s &&
455 abs(filter.x().at<1>() - wind_east) < 0.25 * m / s &&
456 "The wind is found within a quarter of a meter per second.");
457 assert(abs(filter.x().at<2>() - scale) < 0.01 * mp_units::one &&
458 "The airspeed scale factor is found within a percent.");
459 assert(spikes_fused == 0 && "The spikes are rejected.");
460 assert(consistent_rejected <= 2 &&
461 "Nearly every consistent update is fused.");
462
463 return 0;
464}()};
