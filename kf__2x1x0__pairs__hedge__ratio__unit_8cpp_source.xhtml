79 // A 2x1x0 filter, random walk dynamic model, no control, with the price of
80 // the independent stock as an update argument.
81 kalman filter{
82 // The state X is the hedge ratio and intercept: [β, α]. As in the
83 // reference, we start from a null hedge ratio and intercept.
84 state{state_t{ratio{0. * mp_units::one}, price{0. * USD}}},
85 // The filter observes the output Z dependent closing price [$].
86 output<output_t>,
87 // As in the reference, the initial estimate uncertainty P is null: the
88 // first closing prices do not update the state, after which the process
89 // uncertainty accumulates.
90 estimate_uncertainty{kalman_internal::ᴀʙᵀ<state_t, state_t>{}},
91 // The process uncertainty Q: Vw = δ / (1 - δ) with δ = 0.0001, for the
92 // dimensionless hedge ratio and for the intercept in dollars.
93 process_uncertainty{[]() -> auto {
94 using process_uncertainty_t = kalman_internal::ᴀʙᵀ<state_t, state_t>;
95 const double delta{0.0001};
96 process_uncertainty_t value;
97 value.at<0, 0>(delta / (1. - delta) * mp_units::one);
98 value.at<1, 1>(delta / (1. - delta) * USD2);
99 return value;
100 }()},
101 // The output uncertainty R: Ve = 0.001 $^2.
102 output_uncertainty{0.001 * USD2},
103 // The output model H = [x, 1] of the regression changes every day with
104 // the independent closing price x.
105 output_model{[]([[maybe_unused]] const state_t &x,
106 const price &independent) -> output_model_t {
107 output_model_t value;
108 value.at<0>(independent);
109 value.at<1>(1.);
110 return value;
111 }},
112 // The hedge ratio and intercept follow a random walk: their prediction
113 // is their last estimate.
114 transition{[](const state_t &x) -> state_t { return x; }},
115 // The regression forecasts the dependent closing price: βx + α.
116 observation{[](const state_t &x, const price &independent) -> output_t {
117 return output_t{x.at<0>() * independent + x.at<1>()};
118 }},
119 // The filter update uses the independent closing price [$] parameter.
120 update_types<price>,
121 // The filter prediction uses no parameter.
122 prediction_types<>};

124 // Verifies a value at the relative accuracy of its expectation.
125 const auto near{
126 [](const auto &value, const auto &expected, double accuracy) -> bool {
127 return abs(value - expected) < accuracy * abs(expected);
128 }};

130 // The first day only updates the filter, as in the reference: with a null
131 // estimate uncertainty, the gain is null and the state remains unchanged.
132 filter.update(20. * USD, 25. * USD);

137 // Then, run a step of the filter every trading day, predicting the hedge
138 // ratio and intercept of the day and updating them with the closing prices
139 // of the independent and dependent stocks of the day.
140 const auto step{[&filter](price independent, price dependent) -> void {
141 filter.predict();
142 filter.update(independent, dependent);
