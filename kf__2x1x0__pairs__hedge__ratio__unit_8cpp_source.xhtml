93 // A 2x1x0 filter, random walk dynamic model, no control, with the price of
94 // the independent stock as an update argument.
95 kalman filter{
96 // The state X is the hedge ratio and intercept: [β, α]. As in the
97 // reference, we start from a null hedge ratio and intercept.
98 state{state_t{ratio{0. * mp_units::one}, price{0. * USD}}},
99 // The filter observes the output Z dependent closing price [$].
100 output<output_t>,
101 // As in the reference, the initial estimate uncertainty P is null: the
102 // first closing prices do not update the state, after which the process
103 // uncertainty accumulates.
104 estimate_uncertainty{kf::ᴀʙᵀ<state_t, state_t>{}},
105 // The process uncertainty Q: Vw = δ / (1 - δ) with δ = 0.0001, for the
106 // dimensionless hedge ratio and for the intercept in dollars.
107 process_uncertainty{[]() -> auto {
108 using process_uncertainty_t = kf::ᴀʙᵀ<state_t, state_t>;
109 const double delta{0.0001};
110 process_uncertainty_t value;
111 value.at<0, 0>(delta / (1. - delta) * mp_units::one);
112 value.at<1, 1>(delta / (1. - delta) * USD2);
113 return value;
114 }()},
115 // The output uncertainty R: Ve = 0.001 $^2.
116 output_uncertainty{0.001 * USD2},
117 // The output model H = [x, 1] of the regression changes every day with
118 // the independent closing price x.
119 output_model{[]([[maybe_unused]] const state_t &x,
120 const price &independent) -> output_model_t {
121 output_model_t value;
122 value.at<0>(independent);
123 value.at<1>(1.);
124 return value;
125 }},
126 // The hedge ratio and intercept follow a random walk: their prediction
127 // is their last estimate.
128 transition{[](const state_t &x) -> state_t { return x; }},
129 // The regression forecasts the dependent closing price: βx + α.
130 observation{[](const state_t &x, const price &independent) -> output_t {
131 return output_t{x.at<0>() * independent + x.at<1>()};
132 }},
133 // The filter update uses the independent closing price [$] parameter.
134 update_types<price>,
135 // The filter prediction uses no parameter.
136 prediction_types<>};

138 // Verifies a value at the relative accuracy of its expectation.
139 const auto near{
140 [](const auto &value, const auto &expected, double accuracy) -> bool {
141 return abs(value - expected) < accuracy * abs(expected);
142 }};

144 // The first day only updates the filter, as in the reference: with a null
145 // estimate uncertainty, the gain is null and the state remains unchanged.
146 filter.update(20. * USD, 25. * USD);

151 // Then, run a step of the filter every trading day, predicting the hedge
152 // ratio and intercept of the day and updating them with the closing prices
153 // of the independent and dependent stocks of the day.
