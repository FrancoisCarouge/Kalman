114const auto beacon_range{
115 [](const position_t &point, const position_t &beacon) -> length {
116 return (point - beacon).magnitude();
117 }};
118
127const auto beacon_range_jacobian{
128 [](const position_t &point, const position_t &beacon) -> output_model_t {
129 const position_t displacement{point - beacon};
130 const length range{displacement.magnitude()};
131 const auto [north_component, east_component,
132 down_component]{displacement};
133 output_model_t value;
134 value.at<0>((north_component / range).numerical_value_in(mp_units::one) *
135 (mp_units::isq::length / north)[mp_units::one]);
136 value.at<1>((east_component / range).numerical_value_in(mp_units::one) *
137 (mp_units::isq::length / east)[mp_units::one]);
138 value.at<2>((down_component / range).numerical_value_in(mp_units::one) *
139 (mp_units::isq::length / down)[mp_units::one]);
140 return value;
141 }};

205[[maybe_unused]] const auto sample{[] -> int {
206 // The beacons, in the order of their identifiers along the N pattern.
207 const position_t beacons[]{{north(0. * m), east(0. * m), down(-1. * m)},
208 {north(10. * m), east(0. * m), down(-3. * m)},
209 {north(0. * m), east(10. * m), down(-2. * m)},
210 {north(10. * m), east(10. * m), down(-1. * m)}};
211 const position_t receiver{north(3. * m), east(4. * m), down(-0.5 * m)};
212 const auto range_variance{0.1 * 0.1 * m2};
213 const double gate{3.}; // standard deviations
214 const int alignment_ranges{100};
215
216 kalman filter{
217 // The state X: the receiver's position, initially the beacons' centre.
218 state{state_t{north(5. * m), east(5. * m), down(-1.75 * m)}},
219 // The output Z: a range [m].
220 output<output_t>,
221 // The estimate uncertainty P [m^2]:
222 estimate_uncertainty{[]() -> covariance_t {
223 covariance_t value;
224 value.at<0, 0>(25. * m2);
225 value.at<1, 1>(25. * m2);
226 value.at<2, 2>(25. * m2);
227 return value;
228 }()},
229 // The process uncertainty Q [m^2]:
230 process_uncertainty{[]() -> covariance_t {
231 covariance_t value;
232 value.at<0, 0>(1e-4 * m2);
233 value.at<1, 1>(1e-4 * m2);
234 value.at<2, 2>(1e-4 * m2);
235 return value;
236 }()},
237 // The output uncertainty R [m^2]:
238 output_uncertainty{range_variance},
239 // The output model H = ∂h/∂X, for the beacon of the range:
240 output_model{[](const state_t &position,
241 const position_t &beacon) -> output_model_t {
242 return beacon_range_jacobian(position, beacon);
243 }},
244 // The receiver is at rest: its prediction is its last estimate, so no
245 // transition function is declared.
246 // The observation h(X), the range to the beacon:
247 observation{
248 [](const state_t &position, const position_t &beacon) -> output_t {
249 return output_t{beacon_range(position, beacon)};
250 }},
251 // The filter update uses the position of the ranged beacon.
252 update_types<position_t>};
253
254 bool reflection_rejected{false};
255 const int ranges{400};
256 for (int index{0}; index < ranges; ++index) {
257 const position_t &beacon{beacons[index % 4]};
258
259 // Deterministic measurement noise, within one standard deviation, and one
260 // reflection three meters long.
261 length measured{beacon_range(receiver, beacon) +
262 (0.1 * std::sin(12.9898 * index) * m)};
263 const bool reflection{index == 300};
264 if (reflection) {
265 measured += 3. * m;
266 }
267
268 filter.predict();
269
270 // The consistency test of the range, computed outside of the filter.
271 const length innovation{measured - beacon_range(filter.x(), beacon)};
272 const output_model_t jacobian{beacon_range_jacobian(filter.x(), beacon)};
273 // The one-element H P Hᵀ plus the variance R is the variance quantity S.
274 const auto innovation_variance{
275 (jacobian * filter.p() * transposed(jacobian)) + range_variance};
276 const bool consistent{innovation * innovation <
277 gate * gate * innovation_variance};
278
279 if (index < alignment_ranges || consistent) {
280 filter.update(beacon, measured);
281 } else if (reflection) {
282 reflection_rejected = true;
283 }
284 }
285
286 assert(reflection_rejected && "The multipath reflection is rejected.");
287 assert(beacon_range(filter.x(), receiver) < 0.1 * m &&
288 "The receiver position is found within ten centimeters.");
289
290 return 0;
291}()};
