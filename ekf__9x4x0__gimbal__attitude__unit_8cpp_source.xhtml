281const auto state_transition_of{
282 [](const attitude &body_to_ned,
283 const body_velocity_vector &delta_velocity) -> transition_jacobian {
284 transition_jacobian value{transition_jacobian::matrix::Identity()};
285 [&body_to_ned,
286 &value]<std::size_t... Index>(std::index_sequence<Index...>) -> void {
287 (value.at<Index / 3, 6 + (Index % 3)>(
288 transition_jacobian::element<Index / 3, 6 + (Index % 3)>{
289 -body_to_ned(static_cast<Eigen::Index>(Index / 3),
290 static_cast<Eigen::Index>(Index % 3)) *
291 time_step}),
292 ...);
293 }(std::make_index_sequence<9>{});
294 const auto [north, east, down] = to_ned(body_to_ned, delta_velocity);
295 value.at<3, 1>(down.numerical_value_in(m / s) *
296 (north_velocity / east_angle)[m / s / rad]);
297 value.at<3, 2>(-east.numerical_value_in(m / s) *
298 (north_velocity / down_angle)[m / s / rad]);
299 value.at<4, 0>(-down.numerical_value_in(m / s) *
300 (east_velocity / north_angle)[m / s / rad]);
301 value.at<4, 2>(north.numerical_value_in(m / s) *
302 (east_velocity / down_angle)[m / s / rad]);
303 value.at<5, 0>(east.numerical_value_in(m / s) *
304 (down_velocity / north_angle)[m / s / rad]);
305 value.at<5, 1>(-north.numerical_value_in(m / s) *
306 (down_velocity / east_angle)[m / s / rad]);
307 return value;
308 }};

359[[maybe_unused]] const auto sample{[] -> int {
360 const ned_acceleration_vector gravity{Eigen::Vector3d{0., 0., 9.80665} *
361 ned_acceleration[m / s2]};
362 const body_angular_velocity_vector true_bias{roll_rate(0.01 * rad / s),
363 pitch_rate(-0.02 * rad / s),
364 yaw_rate(0.005 * rad / s)};
365 const auto gyroscope_noise{0.001 * rad / s};
366 const auto accelerometer_noise{0.05 * m / s2};
367 // The bias random walk of 1e-4 rad.s^-1 per square root of second, over a
368 // step.
369 const auto bias_noise{1e-4 * rad / (s * sqrt(s)) * sqrt(time_step)};
370
371 kalman filter{
372 // The state X: the errors, zero after each correction.
373 state{error_state{}},
374 // The output Z: the errors of the velocity and of the heading.
375 output<error_measurement>,
376 // The estimate uncertainty P:
377 estimate_uncertainty{[]() -> error_state_covariance {
378 error_state_covariance value;
379 value.at<0, 0>(0.01 * rad2);
380 value.at<1, 1>(0.01 * rad2);
381 value.at<2, 2>(0.01 * rad2);
382 value.at<3, 3>(0.01 * m2 / s2);
383 value.at<4, 4>(0.01 * m2 / s2);
384 value.at<5, 5>(0.01 * m2 / s2);
385 value.at<6, 6>(0.0025 * rad2 / s2);
386 value.at<7, 7>(0.0025 * rad2 / s2);
387 value.at<8, 8>(0.0025 * rad2 / s2);
388 return value;
389 }()},
390 // The process uncertainty Q, of the sensor noises over a step:
391 process_uncertainty{[&]() -> error_state_covariance {
392 const auto attitude_variance{pow<2>(gyroscope_noise * time_step)};
393 const auto velocity_variance{pow<2>(accelerometer_noise * time_step)};
394 error_state_covariance value;
395 value.at<0, 0>(attitude_variance);
396 value.at<1, 1>(attitude_variance);
397 value.at<2, 2>(attitude_variance);
398 value.at<3, 3>(velocity_variance);
399 value.at<4, 4>(velocity_variance);
400 value.at<5, 5>(velocity_variance);
401 value.at<6, 6>(pow<2>(bias_noise));
402 value.at<7, 7>(pow<2>(bias_noise));
403 value.at<8, 8>(pow<2>(bias_noise));
404 return value;
405 }()},
406 // The output uncertainty R:
407 output_uncertainty{[]() -> error_measurement_covariance {
408 error_measurement_covariance value;
409 value.at<0, 0>(0.01 * m2 / s2);
410 value.at<1, 1>(0.01 * m2 / s2);
411 value.at<2, 2>(0.01 * m2 / s2);
412 value.at<3, 3>(0.0025 * rad2);
413 return value;
414 }()},
415 // The output model H selects the observed errors:
416 output_model{[](const error_state &) -> measurement_jacobian {
417 measurement_jacobian value;
418 value.at<0, 3>(1.);
419 value.at<1, 4>(1.);
420 value.at<2, 5>(1.);
421 value.at<3, 2>(1.);
422 return value;
423 }},
424 // The state transition F, of the attitude and the velocity increment:
425 state_transition{state_transition_of},
426 // The observation h(X) of the observed errors:
427 observation{[](const error_state &errors) -> error_measurement {
428 return error_measurement{errors.at<3>(), errors.at<4>(), errors.at<5>(),
429 errors.at<2>()};
430 }},
431 // The filter prediction uses the attitude and the velocity increment.
432 prediction_types<attitude, body_velocity_vector>};
433
434 // The nominal values the errors correct.
435 attitude body_to_ned{rotate_ned(ned_rotation_vector{north_angle(0.02 * rad),
436 east_angle(-0.03 * rad),
437 down_angle(0.1 * rad)},
438 attitude::Identity())};
439 ned_velocity_vector velocity{Eigen::Vector3d{Eigen::Vector3d::Zero()} *
440 ned_velocity[m / s]};
441 body_angular_velocity_vector bias{Eigen::Vector3d{Eigen::Vector3d::Zero()} *
442 body_angular_velocity[rad / s]};
443
444 // Move the estimated errors into the nominal values and reset them.
445 const auto reset{[&filter, &body_to_ned, &velocity, &bias] -> void {
446 const error_state errors{filter.x()};
447 body_to_ned = rotate_ned(
448 ned_rotation_vector{errors.at<0>(), errors.at<1>(), errors.at<2>()},
449 body_to_ned);
450 velocity +=
451 ned_velocity_vector{errors.at<3>(), errors.at<4>(), errors.at<5>()};
452 bias += body_angular_velocity_vector{errors.at<6>(), errors.at<7>(),
453 errors.at<8>()};
454 filter.x(error_state{});
455 }};
456
457 attitude true_body_to_ned{attitude::Identity()};
458 const int steps{1500};
459 for (int step{0}; step < steps; ++step) {
460 const auto time{step * time_step};
461
462 // The true motion: a slow pan with a small oscillation in tilt.
463 const body_angular_velocity_vector true_rate{
464 roll_rate(0.05 * rad / s * mp_units::si::sin(0.3 * rad / s * time)),
465 pitch_rate(0.05 * rad / s * mp_units::si::cos(0.2 * rad / s * time)),
466 yaw_rate(0.1 * rad / s)};
467 const body_rotation_vector delta_angle{(true_rate + true_bias) * time_step};
468 const body_velocity_vector delta_velocity{
469 to_body(true_body_to_ned, -gravity * time_step)};
470 true_body_to_ned = rotate_body(true_body_to_ned, true_rate * time_step);
471
472 // Predict the errors over the step, then integrate the nominal values.
473 filter.predict(body_to_ned, delta_velocity);
474 velocity += to_ned(body_to_ned, delta_velocity) + (gravity * time_step);
475 body_to_ned = rotate_body(body_to_ned, delta_angle - (bias * time_step));
476
477 // Ten times a second, observe the zero velocity of the vehicle at rest,
478 // and the heading of the compass.
479 if (step % 5 == 0) {
480 const heading measured{
481 heading_of(true_body_to_ned) +
482 (0.05 * std::sin(12.9898 * step) * down_angle[rad])};
483 const auto [north, east, down]{velocity};
484 filter.update(-north, -east, -down,
485 wrap(measured - heading_of(body_to_ned)));
486 reset();
487 }
488 }
489
490 assert((bias - true_bias).numerical_value_in(rad / s).norm() < 0.001 &&
491 "The gyroscope bias is found within a milliradian per second.");
492 assert(abs(wrap(heading_of(body_to_ned) - heading_of(true_body_to_ned))) <
493 0.01 * rad &&
494 "The heading is found within ten milliradians.");
495
496 return 0;
497}()};
