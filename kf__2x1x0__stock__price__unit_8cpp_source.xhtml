115[[maybe_unused]] const auto sample{[] -> int {
116 // A 2x1x0 filter, local linear trend dynamic model, no control, constant
117 // daily time step.
118 kalman filter{
119 // The state X is chosen to be the price level and slope: [μ, ν]. We don't
120 // know the stock price; we will set the initial level and slope to 0.
121 state{trend{price{0. * USD}, price_slope{0. * USD / d}}},
122 // The filter observes the output Z closing price [$].
123 output<closing_price>,
124 // Since our initial state is a guess, we will set a very high estimate
125 // uncertainty, an approximation of the diffuse initialization of the
126 // model. The first closing price then sets the level and the second one
127 // sets the slope.
128 estimate_uncertainty{[]() -> auto {
129 trend_covariance value;
130 value.at<0, 0>(1000. * USD2);
131 value.at<1, 1>(1000. * USD2 / d / d);
132 return value;
133 }()},
134 // The process uncertainty Q holds the variances of the level and slope
135 // disturbances: σξ^2 = 0.25 $^2 and σζ^2 = 0.01 $^2/d^2.
136 process_uncertainty{[]() -> auto {
137 trend_covariance value;
138 value.at<0, 0>(0.25 * USD2);
139 value.at<1, 1>(0.01 * USD2 / d / d);
140 return value;
141 }()},
142 // The output uncertainty R is the variance of the transient fluctuations
143 // of the closing price around its level: σε^2 = 1 $^2.
144 output_uncertainty{1. * USD2},
145 // The output model H observes the level only.
146 output_model{[]() -> auto {
147 closing_price_model value;
148 value.at<0>(1.);
149 return value;
150 }()},
151 // The state transition F adds the slope over one day to the level.
152 state_transition{[]() -> auto {
153 trend_transition value;
154 value.at<0, 0>(1.);
155 value.at<0, 1>(1. * d);
156 value.at<1, 1>(1.);
157 return value;
158 }()}};
159
160 // Verifies a value at the relative accuracy of its expectation.
161 const auto near{
162 [](const auto &value, const auto &expected, double accuracy) -> bool {
163 return abs(value - expected) < accuracy * abs(expected);
164 }};
165
166 // Run a step of the filter, updating with the closing price of the day and
167 // predicting the closing price of the next day, every trading day.
168 const auto step{[&filter](price close) -> void {
169 filter.update(close);
170 filter.predict();
171 }};
172
173 step(100.87 * USD);
174
175 // The first closing price sets the level, the slope remains unknown.

178 "The state estimates expected at 0.1% accuracy.");
