62using price = mp_units::quantity<USD>;
63template <typename... Types> using vector = column_vector<double, Types...>;
64using price_slope = mp_units::quantity<USD / d>;
65using state_t = vector<price, price_slope>;
66using output_t = vector<price>;

95[[maybe_unused]] const auto sample{[] -> int {
96 // A 2x1x0 filter, local linear trend dynamic model, no control, constant
97 // daily time step.
98 kalman filter{
99 // The state X is chosen to be the price level and slope: [μ, ν]. We don't
100 // know the stock price; we will set the initial level and slope to 0.
101 state{state_t{price{0. * USD}, price_slope{0. * USD / d}}},
102 // The filter observes the output Z closing price [$].
103 output<output_t>,
104 // Since our initial state is a guess, we will set a very high estimate
105 // uncertainty, an approximation of the diffuse initialization of the
106 // model. The first closing price then sets the level and the second one
107 // sets the slope.
108 estimate_uncertainty{[]() -> auto {
109 using estimate_uncertainty_t = kf::ᴀʙᵀ<state_t, state_t>;
110 estimate_uncertainty_t value;
111 value.at<0, 0>(1000. * USD2);
112 value.at<1, 1>(1000. * USD2 / d / d);
113 return value;
114 }()},
115 // The process uncertainty Q holds the variances of the level and slope
116 // disturbances: σξ^2 = 0.25 $^2 and σζ^2 = 0.01 $^2/d^2.
117 process_uncertainty{[]() -> auto {
118 using process_uncertainty_t = kf::ᴀʙᵀ<state_t, state_t>;
119 process_uncertainty_t value;
120 value.at<0, 0>(0.25 * USD2);
121 value.at<1, 1>(0.01 * USD2 / d / d);
122 return value;
123 }()},
124 // The output uncertainty R is the variance of the transient fluctuations
125 // of the closing price around its level: σε^2 = 1 $^2.
126 output_uncertainty{1. * USD2},
127 // The output model H observes the level only.
128 output_model{[]() -> auto {
129 using output_model_t = kf::evaluate<kf::quotient<output_t, state_t>>;
130 output_model_t value;
131 value.at<0>(1.);
132 return value;
133 }()},
134 // The state transition F adds the slope over one day to the level.
135 state_transition{[]() -> auto {
136 using state_transition_t = kf::evaluate<kf::quotient<state_t, state_t>>;
137 state_transition_t value;
138 value.at<0, 0>(1.);
139 value.at<0, 1>(1. * d);
140 value.at<1, 1>(1.);
141 return value;
142 }()}};
143
144 // Verifies a value at the relative accuracy of its expectation.
145 const auto near{
146 [](const auto &value, const auto &expected, double accuracy) -> bool {
147 return abs(value - expected) < accuracy * abs(expected);
148 }};
149
150 // Run a step of the filter, updating with the closing price of the day and
151 // predicting the closing price of the next day, every trading day.
152 const auto step{[&filter](price close) -> void {
153 filter.update(close);
154 filter.predict();
155 }};
156
157 step(100.87 * USD);
