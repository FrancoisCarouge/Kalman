179const auto output_model_of{[](const track &estimate) -> measurement_jacobian {
180 const length along_x{estimate.at<0>()};
181 const length along_y{estimate.at<1>()};
182 const squared_length squared{(along_x * along_x) + (along_y * along_y)};
183 const length hypotenuse{hypot(along_x, along_y)};
184 measurement_jacobian value;
185 value.at<0, 0>((bearing / x_position)(-along_y / squared));
186 value.at<0, 1>((bearing / y_position)(along_x / squared));
187 value.at<1, 0>((range / x_position)(along_x / hypotenuse));
188 value.at<1, 1>((range / y_position)(along_y / hypotenuse));
189 return value;
190}};

246[[maybe_unused]] const auto sample{[] -> int {
247 const auto radius{8. * m};
248 const auto angular_rate{0.2 * rad / s};
249 const auto bearing_deviation{0.26 * rad};
250 const auto range_deviation{0.3 * m};
251 const auto position_noise{0.03 * m};
252 const auto velocity_noise{0.2 * m / s};
253
254 // The target starts near the negative x axis, at a bearing of nearly π.
255 auto phase{3. * rad};
256 // The seed of the sensor noise, from the system's entropy source: a
257 // blocking input-output operation, out of the real-time verification.
258 const auto seed{[] -> std::random_device::result_type {
259 const not_realtime opt_out;
260 return std::random_device{}();
261 }()};
262 // The sensor noise, a standard normal sequence. The plots are of the seed 1,
263 // with the GNU C++ standard library: the algorithm of the distribution, so
264 // the sequence of a seed, varies across standard libraries.
265 std::minstd_rand engine{seed};
266 std::normal_distribution distribution{0., 1.};
267 const auto noise{[&] -> double { return distribution(engine); }};
268
269 kalman filter{
270 // The state X: the target's position, from its first observation, at
271 // rest.
272 state{track{abscissa{radius * mp_units::si::cos(phase)},
273 ordinate{radius * mp_units::si::sin(phase)},
274 abscissa_rate{0. * m / s}, ordinate_rate{0. * m / s}}},
275 // The output Z: the bearing and the range.
276 output<measurement>,
277 // The estimate uncertainty P:
278 estimate_uncertainty{[]() -> track_covariance {
279 track_covariance value;
280 value.at<0, 0>(25. * m2);
281 value.at<1, 1>(25. * m2);
282 value.at<2, 2>(1. * m2 / s2);
283 value.at<3, 3>(1. * m2 / s2);
284 return value;
285 }()},
286 // The process uncertainty Q, whose velocity noise absorbs the turn:
287 process_uncertainty{[&]() -> track_covariance {
288 track_covariance value;
289 value.at<0, 0>(pow<2>(position_noise));
290 value.at<1, 1>(pow<2>(position_noise));
291 value.at<2, 2>(pow<2>(velocity_noise));
292 value.at<3, 3>(pow<2>(velocity_noise));
293 return value;
294 }()},
295 // The output uncertainty R:
296 output_uncertainty{[&]() -> measurement_covariance {
297 measurement_covariance value;
298 value.at<0, 0>(pow<2>(bearing_deviation));
299 value.at<1, 1>(pow<2>(range_deviation));
300 return value;
301 }()},
302 // The output model H = ∂h/∂X, at the state:
303 output_model{output_model_of},
304 // The constant velocity model: its Jacobian F, and the transition itself.
305 state_transition{[](const track &) -> track_jacobian {
306 track_jacobian value{track_jacobian::matrix::Identity()};
307 value.at<0, 2>(time_step);
308 value.at<1, 3>(time_step);
309 return value;
310 }},
311 transition{[](const track &estimate) -> track {
312 return track{estimate.at<0>() + (estimate.at<2>() * time_step),
313 estimate.at<1>() + (estimate.at<3>() * time_step),
314 estimate.at<2>(), estimate.at<3>()};
315 }},
316 // The observation h(X), the polar coordinates of the position:
317 observation{[](const track &estimate) -> measurement {
318 return polar(estimate.at<0>(), estimate.at<1>());
319 }}};
320
321 const int steps{1000};
322 squared_length squared_errors{0. * m2};
323 int crossings{0};
324 azimuth previous_bearing{wrap(azimuth{phase})};
325 for (int step{0}; step < steps; ++step) {
326 phase += angular_rate * time_step;
327 const abscissa target_x{radius * mp_units::si::cos(phase)};
328 const ordinate target_y{radius * mp_units::si::sin(phase)};
329 const measurement target{polar(target_x, target_y)};
330 const azimuth measured_bearing{
331 wrap(target.at<0>() + (bearing_deviation * noise()))};
332 const distance measured_range{target.at<1>() + (range_deviation * noise())};
333 crossings += static_cast<int>(abs(measured_bearing - previous_bearing) >
334 std::numbers::pi * rad);
335 previous_bearing = measured_bearing;
336
337 filter.predict();
338
339 // Unwrap the measured bearing around the predicted one, so that the
340 // innovation is the angle between them.
341 const azimuth predicted_bearing{
342 polar(filter.x().at<0>(), filter.x().at<1>()).at<0>()};
343 filter.update(predicted_bearing +
344 wrap(measured_bearing - predicted_bearing),
345 measured_range);
346
347 squared_errors += pow<2>(length{filter.x().at<0>() - target_x}) +
348 pow<2>(length{filter.x().at<1>() - target_y});
349 }
350
351 // The target crossed the negative x axis four times, the measured bearing
352 // jumping from π to -π at each crossing, and the estimate stayed close to the
353 // target, whatever the seed of the noise.
354 assert(crossings >= 4 &&
355 "The measured bearing jumps at each crossing of the negative x axis.");
356 assert(sqrt(squared_errors / steps) < 1. * m &&
357 "The estimated position stays near the target.");
358
359 return 0;
360}()};
