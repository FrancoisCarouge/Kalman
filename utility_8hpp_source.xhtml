553constexpr void for_constexpr(Function &&function) {
554 if constexpr (Begin < End) {
555 function(std::integral_constant<std::size_t, Begin>());
556 for_constexpr<Begin + Increment, End, Increment>(
557 std::forward<Function>(function));
558 }

568template <typename Callable>
569constexpr void assign(Callable &callable, const auto &...values) {
570 if constexpr (std::is_move_assignable_v<Callable>) {
571 callable = Callable{values...};
572 } else {
573 static_assert(std::is_nothrow_move_constructible_v<Callable>,
574 "The callable characteristic must be move assignable or "
575 "non-throwing move constructible to be replaced.");
576
577 Callable replacement{values...};
578 std::destroy_at(&callable);
579 std::construct_at(&callable, std::move(replacement));
580 }
581}
