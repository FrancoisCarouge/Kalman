618template <typename Callable>
619constexpr void assign(Callable &callable, const auto &...values) {
620 if constexpr (std::is_move_assignable_v<Callable>) {
621 callable = Callable{values...};
622 } else {
623 static_assert(std::is_nothrow_move_constructible_v<Callable>,
624 "The callable characteristic must be move assignable or "
625 "non-throwing move constructible to be replaced.");
626
627 Callable replacement{values...};
628 std::destroy_at(&callable);
629 std::construct_at(&callable, std::move(replacement));
630 }
631}
