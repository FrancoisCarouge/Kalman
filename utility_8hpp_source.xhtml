539template <std::size_t Begin, std::size_t End, std::size_t Increment,
540 typename Function>
541constexpr void for_constexpr(Function &&function) {
542 if constexpr (Begin < End) {
543 function(std::integral_constant<std::size_t, Begin>());
544 for_constexpr<Begin + Increment, End, Increment>(
545 std::forward<Function>(function));
546 }
547}

556template <typename Callable>
557constexpr void assign(Callable &callable, const auto &...values) {
558 if constexpr (std::is_move_assignable_v<Callable>) {
559 callable = Callable{values...};
560 } else {
561 static_assert(std::is_nothrow_move_constructible_v<Callable>,
562 "The callable characteristic must be move assignable or "
563 "non-throwing move constructible to be replaced.");
564
565 Callable replacement{values...};
566 std::destroy_at(&callable);
567 std::construct_at(&callable, std::move(replacement));
568 }
569}
