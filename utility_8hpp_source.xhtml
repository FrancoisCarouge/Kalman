83template <typename Type, std::size_t Position>
84[[nodiscard]] consteval auto deduce_element() {
85 if constexpr (requires(const Type value) { value.template at<Position>(); }) {
86 return std::type_identity<std::remove_cvref_t<
87 decltype(std::declval<const Type &>().template at<Position>())>>{};
88 } else if constexpr (requires(const Type value) { value(Position); }) {
89 return std::type_identity<std::remove_cvref_t<
90 decltype(std::declval<const Type &>()(Position))>>{};
91 } else {
92 return std::type_identity<void>{};
93 }

661template <std::size_t Begin, std::size_t End, std::size_t Increment,
662 typename Function>
663constexpr void for_constexpr(Function &&function) {
664 if constexpr (Begin < End) {
665 function(std::integral_constant<std::size_t, Begin>{});
666 for_constexpr<Begin + Increment, End, Increment>(
667 std::forward<Function>(function));
668 }

678template <typename Callable>
679constexpr void assign(Callable &callable, const auto &...values) {
680 if constexpr (std::is_move_assignable_v<Callable>) {
681 callable = Callable{values...};
682 } else {
683 static_assert(std::is_nothrow_move_constructible_v<Callable>,
684 "The callable characteristic must be move assignable or "
685 "non-throwing move constructible to be replaced.");
686
687 Callable replacement{values...};
688 std::destroy_at(&callable);
689 std::construct_at(&callable, std::move(replacement));
690 }
691}
