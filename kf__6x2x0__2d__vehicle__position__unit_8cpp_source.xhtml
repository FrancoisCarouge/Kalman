121[[maybe_unused]] const auto sample{[] -> int {
122 // A 6x2x0 filter, constant acceleration dynamic model, no control.
123 kalman filter{
124 // The state X is chosen to be the position, velocity, acceleration in the
125 // XY plane: [px, vx, ax, py, vy, ay]. We don't know the vehicle location;
126 // we will set initial position, velocity and acceleration to 0.
127 state{motion{position{0. * m}, velocity{0. * m / s},
128 acceleration{0. * m / s2}, position{0. * m},
129 velocity{0. * m / s}, acceleration{0. * m / s2}}},
130 // The vehicle has an onboard location sensor that reports output Z as X
131 // and Y coordinates of the system.
132 output<location>,
133 // The estimate uncertainty matrix P.
134 // Since our initial state vector is a guess, we will set a very high
135 // estimate uncertainty. The high estimate uncertainty results in a high
136 // Kalman Gain, giving a high weight to the measurement.
137 estimate_uncertainty{[]() -> auto {
138 motion_covariance value;
139 value.at<0, 0>(500. * m2);
140 value.at<1, 1>(500. * m2 / s2);
141 value.at<2, 2>(500. * m2 / s4);
142 value.at<3, 3>(500. * m2);
143 value.at<4, 4>(500. * m2 / s2);
144 value.at<5, 5>(500. * m2 / s4);
145 return value;
146 }()},
147 // The process uncertainty noise matrix Q, constant, computed in place,
148 // with random acceleration standard deviation: σa = 0.2 m.s^-2.
149 process_uncertainty{[]() -> auto {
150 motion_covariance value;
151 value.at<0, 0>(0.2 * 0.2 * 0.25 * m2);
152 value.at<0, 1>(0.2 * 0.2 * 0.5 * m2 / s);
153 value.at<0, 2>(0.2 * 0.2 * 0.5 * m2 / s2);
154 value.at<1, 0>(0.2 * 0.2 * 0.5 * m2 / s);
155 value.at<1, 1>(0.2 * 0.2 * m2 / s2);
156 value.at<1, 2>(0.2 * 0.2 * m2 / s3);
157 value.at<2, 0>(0.2 * 0.2 * 0.5 * m2 / s2);
158 value.at<2, 1>(0.2 * 0.2 * m2 / s3);
159 value.at<2, 2>(0.2 * 0.2 * m2 / s4);
160 value.at<3, 3>(0.2 * 0.2 * 0.25 * m2);
161 value.at<3, 4>(0.2 * 0.2 * 0.5 * m2 / s);
162 value.at<3, 5>(0.2 * 0.2 * 0.5 * m2 / s2);
163 value.at<4, 3>(0.2 * 0.2 * 0.5 * m2 / s);
164 value.at<4, 4>(0.2 * 0.2 * m2 / s2);
165 value.at<4, 5>(0.2 * 0.2 * m2 / s3);
166 value.at<5, 3>(0.2 * 0.2 * 0.5 * m2 / s2);
167 value.at<5, 4>(0.2 * 0.2 * m2 / s3);
168 value.at<5, 5>(0.2 * 0.2 * m2 / s4);
169 return value;
170 }()},
171 // The output uncertainty matrix R. Assume that the x and y measurements
172 // are uncorrelated, i.e. error in the x coordinate measurement doesn't
173 // depend on the error in the y coordinate measurement. In real-life
174 // applications, the measurement uncertainty can differ between
175 // measurements. In many systems the measurement uncertainty depends on
176 // the measurement SNR (signal-to-noise ratio), angle between sensor (or
177 // sensors) and target, signal frequency and many other parameters.
178 // For the sake of the example simplicity, we will assume a constant
179 // measurement uncertainty: R1 = R2...Rn-1 = Rn = R The measurement error
180 // standard deviation: σxm = σym = 3m. The variance 9.
181 output_uncertainty{{9. * m2, 0. * m2}, {0. * m2, 9. * m2}},
182 // The output model matrix H. The dimension of zn is 2x1 and the dimension
183 // of xn is 6x1. Therefore the dimension of the observation matrix H shall
184 // be 2x6.
185 output_model{[]() -> auto {
186 location_model value;
187 value.at<0, 0>(1.);
188 value.at<1, 3>(1.);
189 return value;
190 }()},
191 // The state transition matrix F would be:
192 state_transition{[]() -> auto {
193 motion_transition value;
194 value.at<0, 0>(1.);
195 value.at<0, 1>(1. * s);
196 value.at<0, 2>(0.5 * s2);
197 value.at<1, 1>(1.);
198 value.at<1, 2>(1. * s);
199 value.at<2, 2>(1.);
200 value.at<3, 3>(1.);
201 value.at<3, 4>(1. * s);
202 value.at<3, 5>(0.5 * s2);
203 value.at<4, 4>(1.);
204 value.at<4, 5>(1. * s);
205 value.at<5, 5>(1.);
206 return value;
207 }()}};

209 // Now we can predict the next state based on the initialization values.
210 filter.predict();
211
212 // The measurement values: z1 = [-393.66 m, 300.4 m]
213 filter.update(-393.66 * m, 300.4 * m);
214 filter.predict();
215 filter.update(-375.93 * m, 301.78 * m);
216 filter.predict();
217
218 // Verify the example estimated state at 0.1% accuracy.
219 assert(position{-277.8 * m} < filter.x().at<0>() &&
220 filter.x().at<0>() < position{-277.7 * m} &&
221 velocity{148.3 * m / s} < filter.x().at<1>() &&
222 filter.x().at<1>() < velocity{148.4 * m / s} &&
223 acceleration{94.5 * m / s2} < filter.x().at<2>() &&
224 filter.x().at<2>() < acceleration{94.56 * m / s2} &&
225 position{249.7 * m} < filter.x().at<3>() &&
226 filter.x().at<3>() < position{249.8 * m} &&
227 velocity{-86.0 * m / s} < filter.x().at<4>() &&
228 filter.x().at<4>() < velocity{-85.9 * m / s} &&
229 acceleration{-63.65 * m / s2} < filter.x().at<5>() &&
230 filter.x().at<5>() < acceleration{-63.64 * m / s2} &&
231 "The state estimates expected at 0.1% accuracy.");

235
236 return 0;
