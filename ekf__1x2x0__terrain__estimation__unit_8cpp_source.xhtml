122const auto predict_range{[](const altitude &terrain, const altitude &vehicle,
123 const tilt_angle &tilt) -> slant_range {
124 return (vehicle - terrain) / mp_units::si::cos(tilt);
125}};
126
131const auto range_jacobian{[](const tilt_angle &tilt) -> range_derivative {
132 return -1. / mp_units::si::cos(tilt);
133}};
134
142const auto predict_flow{[](const altitude &terrain, const altitude &vehicle,
143 const tilt_angle &tilt,
144 const ground_speed &speed) -> optical_flow {
145 return flow_rate(speed / predict_range(terrain, vehicle, tilt));
146}};
147
153const auto flow_jacobian{[](const altitude &terrain, const altitude &vehicle,
154 const tilt_angle &tilt,
155 const ground_speed &speed) -> flow_derivative {
156 return predict_flow(terrain, vehicle, tilt, speed) / (vehicle - terrain);
157}};

226[[maybe_unused]] const auto sample{[] -> int {
227 const auto time_step{0.1 * s};
228 const ground_speed speed{6. * m / s};
229 const altitude vehicle{30. * m};
230 // The ground rises north, at a gradient under an assumed maximum.
231 const gradient terrain_slope{0.03 * gradient::reference};
232 const gradient terrain_gradient_maximum{0.2 * gradient::reference};
233 const auto range_deviation{0.05 * m};
234 const auto flow_deviation{0.01 * rad / s};
235 const height ground_clearance{0.1 * m};
236 const double gate{3.}; // standard deviations
237
238 // The terrain change over a step at the maximum gradient.
239 const altitude terrain_change{terrain_gradient_maximum * speed * time_step};
240
241 // The true terrain altitude below the vehicle at a time.
242 const auto true_terrain{[&](const auto &time) -> altitude {
243 return 2. * m + (terrain_slope * speed * time);
244 }};
245
246 kalman filter{
247 // The state X: the terrain altitude, initially at the take-off point.
248 state{terrain_altitude{altitude{0. * m}}},
249 // The output Z: the range [m] and the flow rate [rad.s^-1].
250 output<measurement>,
251 // The estimate uncertainty P [m^2]:
252 estimate_uncertainty{variance{100. * m2}},
253 // The process uncertainty Q [m^2]:
254 process_uncertainty{variance{terrain_change * terrain_change}},
255 // The output uncertainty R, of independent sensors:
256 output_uncertainty{[&]() -> measurement_covariance {
257 measurement_covariance value;
258 value.at<0, 0>(range_deviation * range_deviation);
259 value.at<1, 1>(flow_deviation * flow_deviation);
260 return value;
261 }()},
262 // The output model H = ∂h/∂X, of the observed sensors:
263 output_model{[](const terrain_altitude &terrain,
264 const altitude &vehicle_altitude, const tilt_angle &tilt,
265 const ground_speed &vehicle_speed,
266 const observed &sensors) -> measurement_jacobian {
267 measurement_jacobian value;
268 if (sensors.range) {
269 value.at<0>(range_jacobian(tilt));
270 }
271 if (sensors.flow) {
272 value.at<1>(flow_jacobian(terrain.at<>(), vehicle_altitude, tilt,
273 vehicle_speed));
274 }
275 return value;
276 }},
277 // The terrain is constant: its prediction is its last estimate, so no
278 // transition function is declared.
279 // The observation h(X), the range and the flow rate:
280 observation{[](const terrain_altitude &terrain,
281 const altitude &vehicle_altitude, const tilt_angle &tilt,
282 const ground_speed &vehicle_speed,
283 const observed &) -> measurement {
284 return measurement{
285 predict_range(terrain.at<>(), vehicle_altitude, tilt),
286 predict_flow(terrain.at<>(), vehicle_altitude, tilt,
287 vehicle_speed)};
288 }},
289 // The filter update uses the vehicle altitude, tilt, and speed, and the
290 // observed sensors.
291 update_types<altitude, tilt_angle, ground_speed, observed>};
292
293 // The consistency test of an output, its innovation within the gate,
294 // computed outside of the filter.
295 const auto consistent{[gate, &filter](const auto &innovation,
296 const auto &derivative,
297 const auto &output_variance) -> bool {
298 // The one-element H P Hᵀ plus R is the innovation variance S.
299 const auto innovation_variance{(derivative * filter.p() * derivative) +
300 output_variance};
301 return innovation * innovation < gate * gate * innovation_variance;
302 }};
303
304 bool branch_rejected{false};
305 const int steps{600};
306 for (int step{0}; step < steps; ++step) {
307 const auto time{step * time_step};
308 const tilt_angle tilt{0.05 * rad * mp_units::si::sin(0.5 * rad / s * time)};
309 const altitude terrain{true_terrain(time)};
310
311 filter.predict();
312
313 // Deterministic measurement noise, within one standard deviation.
314 const slant_range range_noise{std::sin(12.9898 * step) * range_deviation};
315 const optical_flow flow_noise{std::sin(78.233 * step) * flow_deviation};
316
317 // The range finder drops out between 20 s and 40 s, and a branch in its
318 // beam returns a range eight meters short at 50 s.
319 const bool range_available{time < 20. * s || time >= 40. * s};
320 const bool branch{step == 500};
321 slant_range range{predict_range(terrain, vehicle, tilt) + range_noise};
322 if (branch) {
323 range -= 8. * m;
324 }
325 const optical_flow flow{predict_flow(terrain, vehicle, tilt, speed) +
326 flow_noise};
327
328 // Observe each available and consistent sensor.
329 const altitude estimate{filter.x().at<>()};
330 const observed sensors{
331 .range =
332 range_available &&
333 consistent(range - predict_range(estimate, vehicle, tilt),
334 range_jacobian(tilt), range_deviation * range_deviation),
335 .flow = consistent(flow - predict_flow(estimate, vehicle, tilt, speed),
336 flow_jacobian(estimate, vehicle, tilt, speed),
337 flow_deviation * flow_deviation)};
338 branch_rejected = branch_rejected || (branch && !sensors.range);
339
340 // Fuse the observed sensors at once, then keep the terrain below the
341 // vehicle.
342 filter.update(vehicle, tilt, speed, sensors, range, flow);
343 filter.x(terrain_altitude{
344 std::min(filter.x().at<>(), vehicle - ground_clearance)});
345
346 // The flow keeps the estimate within half a meter through the dropout.
347 if (!range_available) {
348 assert(abs(filter.x().at<>() - terrain) < 0.5 * m &&
349 "The optical flow alone keeps the terrain estimate.");
350 assert(filter.p() < 0.16 * m2 &&
351 "The optical flow bounds the terrain standard deviation within "
352 "0.4 m, which the process noise alone would grow to 1.7 m.");
353 }
354 }
355
356 assert(branch_rejected && "The short range of the branch is rejected.");
357 assert(abs(filter.x().at<>() - true_terrain((steps - 1) * time_step)) <
358 0.2 * m &&
359 "The range finder brings the terrain estimate back.");
360
361 return 0;
362}()};
